cmake_minimum_required(VERSION 3.22.1)
project(volkino)
if(NOT VOLKINO_NODE_DIR)
    message(FATAL_ERROR "VOLKINO_NODE_DIR must select the Node.js runtime for this APK flavor")
endif()
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION ${VOLKINO_NODE_DIR}/bin/${ANDROID_ABI}/libnode.so)
add_library(volkino SHARED runtime.cpp)
target_include_directories(volkino PRIVATE ${VOLKINO_NODE_DIR}/include/node)
target_link_libraries(volkino node log)
